"""#335 AC-4: integrity of the DB copy in the stock-data volume after the daily run."""

import json
import sqlite3

con = sqlite3.connect("file:/data/stock.db?mode=ro", uri=True)
print("PRAGMA integrity_check:", [r[0] for r in con.execute("PRAGMA integrity_check")])
print("PRAGMA foreign_key_check rows:", len(con.execute("PRAGMA foreign_key_check").fetchall()))
print("alembic_version:", con.execute("select version_num from alembic_version").fetchone()[0])
cols = [r[1] for r in con.execute("PRAGMA table_info(schedule_runs)")]
for row in con.execute("select * from schedule_runs order by id desc limit 2"):
    rec = dict(zip(cols, row))
    summary = json.loads(rec.get("summary_json") or "{}")
    keep = {k: rec[k] for k in cols if k not in ("summary_json",)}
    keep["runtime"] = summary.get("runtime")
    print("schedule_runs:", json.dumps(keep, ensure_ascii=False, default=str))
steps = [r[1] for r in con.execute("PRAGMA table_info(schedule_run_steps)")]
last = con.execute("select max(id) from schedule_runs").fetchone()[0]
key = "run_id" if "run_id" in steps else "schedule_run_id"
for row in con.execute(f"select * from schedule_run_steps where {key} = ? order by id", (last,)):
    rec = dict(zip(steps, row))
    print("step:", rec.get("step") or rec.get("name"), rec.get("status"), (rec.get("error") or "")[:120])
